
# queries/domain_enforcement.yaml
```yaml
---
- id: KICS-TF-001
  category: DOMAIN_ENFORCEMENT
  severity: HIGH
  description: >
    AWS IAM resources (aws_iam_*) must be defined only in files matching *iam*.tf
  query: |
    SELECT
      metadata.file_path AS file,
      resource.resource_name AS resource
    FROM
      terraform_resource AS resource,
      metadata AS metadata
    WHERE
      resource.resource_type LIKE 'aws_iam_%'
      AND NOT metadata.file_path REGEXP '.*iam.*\.tf$'

- id: KICS-TF-002
  category: DOMAIN_ENFORCEMENT
  severity: HIGH
  description: >
    Azure IAM resources (azurerm_role_*) must be defined only in files matching *iam*.tf
  query: |
    SELECT
      metadata.file_path AS file,
      resource.resource_name AS resource
    FROM
      terraform_resource AS resource,
      metadata AS metadata
    WHERE
      resource.resource_type LIKE 'azurerm_role_%'
      AND NOT metadata.file_path REGEXP '.*iam.*\.tf$'
```
